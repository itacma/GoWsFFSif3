-- ALTER SESSION SET CURRENT_SCHEMA = FF_CONT;

-- 1. TABLA: LISTA_OFERTAS_VENTA
CREATE TABLE FF_CONT.LISTA_OFERTAS_VENTA (
    LISTA_OFT_ID  NUMBER(6) GENERATED ALWAYS AS IDENTITY MINVALUE 1 MAXVALUE 999999 INCREMENT BY 1 START WITH 1 CACHE 20 NOT NULL,
    NOMBRE        VARCHAR2(150) NOT NULL,
    VIEW_PRECIOS  VARCHAR2(50) NOT NULL,
    CONSTRAINT PK_LISTA_OFERTAS_VENTA PRIMARY KEY (LISTA_OFT_ID)
);

-- 2. TABLA: LISTA_OFERTAS_VENTA_SUCURSAL
CREATE TABLE FF_CONT.LISTA_OFERTAS_VENTA_SUCURSAL (
    LISTA_OFT_ID  NUMBER(6) NOT NULL,
    SUCURSAL_ID   NUMBER NOT NULL,
    CONSTRAINT PK_LISTA_OFT_SUCURSAL PRIMARY KEY (LISTA_OFT_ID, SUCURSAL_ID),
    CONSTRAINT FK_LOVS_LISTA_OFT FOREIGN KEY (LISTA_OFT_ID) 
        REFERENCES FF_CONT.LISTA_OFERTAS_VENTA (LISTA_OFT_ID) ON DELETE CASCADE
);

-- 3. TABLA: LISTA_REQ_CAMPO
CREATE TABLE FF_CONT.LISTA_REQ_CAMPO (
    LISTA_REQ_CAMPO_ID  NUMBER GENERATED BY DEFAULT AS IDENTITY START WITH 1 INCREMENT BY 1 NOT NULL,
    CAMPO_NOMBRE        VARCHAR2(50), -- [NOMBRE, VIGENCIA, TIPO]
    CAMPO_TIPO          VARCHAR2(20), -- [STRING, DATETIME, COMBOBOX]
    REQUERIDO           CHAR(1),      -- 'S' o 'N'
    CONSTRAINT PK_LISTA_REQ_CAMPO PRIMARY KEY (LISTA_REQ_CAMPO_ID),
    CONSTRAINT CK_LRC_REQUERIDO CHECK (REQUERIDO IN ('S', 'N'))
);

-- 4. TABLA: LISTA_OFT_SKU_REQ
CREATE TABLE FF_CONT.LISTA_OFT_SKU_REQ (
    LISTA_OFT_SKU_REQ_ID  NUMBER GENERATED BY DEFAULT AS IDENTITY START WITH 1 INCREMENT BY 1 NOT NULL,
    ARTICULO_ID           VARCHAR2(100) NOT NULL,
    CANTIDAD              NUMBER NOT NULL,
    CONSTRAINT PK_LISTA_OFT_SKU_REQ PRIMARY KEY (LISTA_OFT_SKU_REQ_ID)
);

-- 5. TABLA: LISTA_OFT_SKU_BNF
CREATE TABLE FF_CONT.LISTA_OFT_SKU_BNF (
    LISTA_OFT_SKU_BNF_ID  NUMBER GENERATED BY DEFAULT AS IDENTITY START WITH 1 INCREMENT BY 1 NOT NULL,
    ARTICULO_ID           VARCHAR2(100) NOT NULL,
    CANTIDAD              NUMBER,
    CONSTRAINT PK_LISTA_OFT_SKU_BNF PRIMARY KEY (LISTA_OFT_SKU_BNF_ID)
);

-- 6. TABLA: LISTA_OFERTAS_PKG
CREATE TABLE FF_CONT.LISTA_OFERTAS_PKG (
    LISTA_OFT_PKG_ID            NUMBER GENERATED BY DEFAULT AS IDENTITY START WITH 1 INCREMENT BY 1 NOT NULL,
    NOMBRE                      VARCHAR2(50) NOT NULL,
    LISTA_OFT_SKU_REQ_ID        NUMBER NOT NULL,
    CANTIDAD_REQ_TOT            NUMBER NOT NULL,
    LISTA_OFT_SKU_BNF_ID        NUMBER NOT NULL,
    IMPORTE_MINIMO_COMPRA       NUMBER(10,4) NOT NULL,
    LOGICA_NEGOCIO_BNF          CHAR(1) NOT NULL, -- '&' o '|'
    LOGICA_NEGOCIO_BNF_PRECIO   CHAR(1) NOT NULL, -- '0' o '1'
    CONSTRAINT PK_LISTA_OFERTAS_PKG PRIMARY KEY (LISTA_OFT_PKG_ID),
    CONSTRAINT FK_LOP_SKU_REQ FOREIGN KEY (LISTA_OFT_SKU_REQ_ID) 
        REFERENCES FF_CONT.LISTA_OFT_SKU_REQ (LISTA_OFT_SKU_REQ_ID),
    CONSTRAINT FK_LOP_SKU_BNF FOREIGN KEY (LISTA_OFT_SKU_BNF_ID) 
        REFERENCES FF_CONT.LISTA_OFT_SKU_BNF (LISTA_OFT_SKU_BNF_ID),
    CONSTRAINT CK_LOP_LOGICA CHECK (LOGICA_NEGOCIO_BNF IN ('&', '|')),
    CONSTRAINT CK_LOP_PRECIO CHECK (LOGICA_NEGOCIO_BNF_PRECIO IN ('0', '1'))
);

drop table LISTA_OFERTAS_VENTA_DET;
-- 7. TABLA: LISTA_OFERTAS_VENTA_DET

CREATE TABLE FF_CONT.LISTA_OFERTAS_VENTA_DET (
    LISTA_OFT_ID        NUMBER(6) NOT NULL,
    ARTICULO_ID         VARCHAR2(40) NOT NULL,
    PARTIDA             NUMBER(2) NOT NULL,
    FECHA_INICIO        DATE,
    FECHA_FIN           DATE,
    CANTIDAD            NUMBER(4) NOT NULL,
    BENEFICIO_CANTIDAD  NUMBER(4) NOT NULL,
    DESCUENTO           NUMBER(5,2) NOT NULL,
    ACTIVO              CHAR(1) NOT NULL, -- 'S' o 'N'
    IMPORTE_BENEFICIO   NUMBER(15,4) NOT NULL,
    IMPORTE_PACTO       NUMBER(15,4) NOT NULL,
    LISTA_OFT_PKG_ID    NUMBER,
    LISTA_REQ_CAMPO_ID  NUMBER,
    CONSTRAINT PK_LISTA_OFERTAS_VENTA_DET PRIMARY KEY (LISTA_OFT_ID, ARTICULO_ID, PARTIDA),
    CONSTRAINT FK_LOVD_LISTA_OFT FOREIGN KEY (LISTA_OFT_ID) 
        REFERENCES FF_CONT.LISTA_OFERTAS_VENTA (LISTA_OFT_ID),
    /*CONSTRAINT FK_LOVD_PKG FOREIGN KEY (LISTA_OFT_PKG_ID) 
        REFERENCES FF_CONT.LISTA_OFERTAS_PKG (LISTA_OFT_PKG_ID),
    CONSTRAINT FK_LOVD_REQ_CAMPO FOREIGN KEY (LISTA_REQ_CAMPO_ID) 
        REFERENCES FF_CONT.LISTA_REQ_CAMPO (LISTA_REQ_CAMPO_ID),*/
    CONSTRAINT CK_LOVD_ACTIVO CHECK (ACTIVO IN ('S', 'N'))
);

drop table LISTA_OFERTAS_CUPONES;
-- 8. TABLA: LISTA_OFERTAS_CUPONES
CREATE TABLE FF_CONT.LISTA_OFERTAS_CUPONES (
    LISTA_CUPONES_ID   NUMBER GENERATED BY DEFAULT AS IDENTITY START WITH 1 INCREMENT BY 1 NOT NULL,
    LISTA_OFT_ID       NUMBER(6) NOT NULL,
    FOLIO_CUPON        NUMBER(6) NOT NULL,
    ACTIVO             CHAR(1) NOT NULL, -- 'S'=Disponible, 'G'=Generada, 'P'=Pagada
    ULT_ACTUALIZACION  DATE,
    REFERENCIA         VARCHAR2(20),
    CONSTRAINT PK_LISTA_OFERTAS_CUPONES PRIMARY KEY (LISTA_CUPONES_ID),
    CONSTRAINT FK_LOC_LISTA_OFT FOREIGN KEY (LISTA_OFT_ID) 
        REFERENCES FF_CONT.LISTA_OFERTAS_VENTA (LISTA_OFT_ID),
    CONSTRAINT CK_LOC_ACTIVO CHECK (ACTIVO IN ('S', 'G', 'P'))
);


-- Índice para acelerar la validación e historial de cupones por folio (api/cupon-vigente)
CREATE INDEX FF_CONT.IDX_CUPONES_FOLIO ON FF_CONT.LISTA_OFERTAS_CUPONES (FOLIO_CUPON, ACTIVO);

-- Índice compuesto para la consulta masiva de tasación por artículo, sucursal y rango de vigencia (api/promocionesVenta)
CREATE INDEX FF_CONT.IDX_LOVD_BUSQUEDA ON FF_CONT.LISTA_OFERTAS_VENTA_DET (ARTICULO_ID, ACTIVO, FECHA_INICIO, FECHA_FIN);
